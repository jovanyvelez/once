Introducción

¿Qué es un menú en programación?

Un menú funcional es una herramienta que organiza las opciones de un programa de forma clara, permitiendo que el usuario elija qué hacer en cada momento. Es como el menú de una aplicación, pero construido con código.

╔══════════════════════════╗ ║ MENÚ PRINCIPAL ║ ╠══════════════════════════╣ ║ 1. Opción A ║ ║ 2. Opción B ║ ║ 3. Opción C ║ ║ 0. Salir ║ ╚══════════════════════════╝

🧩 Las tres partes de todo menú

Todo menú bien construido sigue la misma estructura interna:

Parte ¿Qué hace? ¿Cómo se construye?
1. Mostrar Imprime las opciones en pantalla Console.WriteLine
2. Capturar Lee la elección del usuario Console.ReadLine
3. Ejecutar Llama a la función correcta según la opción switch o if/else

Bloque 1 — Fundamentos del Menú

Tu primer programa interactivo
⏱️ 45 min 📚 2 conceptos

🔄 Concepto 1.1 — El do...while como motor del menú

El ciclo do...while es ideal para menús porque ejecuta el menú al menos una vez antes de preguntar si debe repetirse.

Flujo do...while
1
Mostrar el menú en pantalla
2
Leer la opción elegida por el usuario
3
Ejecutar la acción correspondiente
¿Opción ≠ salir?
🔁 Volver al paso 1
¿Opción = salir?
✅ Terminar el programa

📝 Ejemplo 1 — El menú más simple

MenuSimple.cs 
using System;

class MenuSimple
{
    static void Main()
    {
        string opcion;

        do
        {
            // ── MOSTRAR ──────────────────────────────
            Console.Clear();
            Console.WriteLine("╔══════════════════════╗");
            Console.WriteLine("║    MI PRIMER MENÚ    ║");
            Console.WriteLine("╠══════════════════════╣");
            Console.WriteLine("║  1. Saludar          ║");
            Console.WriteLine("║  2. Despedirse       ║");
            Console.WriteLine("║  0. Salir            ║");
            Console.WriteLine("╚══════════════════════╝");
            Console.Write("Elige una opción: ");

            // ── CAPTURAR ─────────────────────────────
            opcion = Console.ReadLine();

            // ── EJECUTAR ─────────────────────────────
            switch (opcion)
            {
                case "1":
                    Console.WriteLine("\n👋 ¡Hola, mundo!");
                    break;
                case "2":
                    Console.WriteLine("\n👋 ¡Hasta luego!");
                    break;
                case "0":
                    Console.WriteLine("\n🚪 Cerrando programa...");
                    break;
                default:
                    Console.WriteLine("\n⚠️  Opción no válida.");
                    break;
            }

            if (opcion != "0")
            {
                Console.Write("\nPresiona ENTER para continuar...");
                Console.ReadLine();
            }

        } while (opcion != "0");
    }
}

🧱 Concepto 1.2 — Separar responsabilidades con funciones

Un error muy común es poner todo el código dentro del Main. Esto hace el programa difícil de leer y de corregir. La solución es usar funciones.

❌ Sin funciones
Main()
  mostrar menú...
  leer opción...
  if opcion == 1
    Console.Write...
    Console.Write...
  if opcion == 2
    ...cien líneas...
✅ Con funciones
Main()
  MostrarMenu()
  LeerOpcion()
  EjecutarOpcion()

  Funcion1()
  Funcion2()
  Funcion3()

📝 Ejemplo 2 — Menú con funciones separadas

MenuConFunciones.cs 
using System;

class MenuConFunciones
{
    // ════════════════════════════════════════════
    //  SECCIÓN 1: FUNCIÓN PRINCIPAL
    // ════════════════════════════════════════════
    static void Main()
    {
        string opcion;

        do
        {
            MostrarMenu();
            opcion = Console.ReadLine();
            EjecutarOpcion(opcion);
        } while (opcion != "0");
    }

    // ════════════════════════════════════════════
    //  SECCIÓN 2: FUNCIONES DE MENÚ
    // ════════════════════════════════════════════
    static void MostrarMenu()
    {
        Console.Clear();
        Console.WriteLine("══════════════════════════");
        Console.WriteLine("       CALCULADORA        ");
        Console.WriteLine("══════════════════════════");
        Console.WriteLine("  1. Sumar dos números");
        Console.WriteLine("  2. Restar dos números");
        Console.WriteLine("  3. Multiplicar");
        Console.WriteLine("  0. Salir");
        Console.WriteLine("══════════════════════════");
        Console.Write("  Opción: ");
    }

    static void EjecutarOpcion(string opcion)
    {
        switch (opcion)
        {
            case "1": Sumar();       break;
            case "2": Restar();      break;
            case "3": Multiplicar(); break;
            case "0": MensajeSalida(); break;
            default:  MensajeError();  break;
        }
        if (opcion != "0") Pausa();
    }

    // ════════════════════════════════════════════
    //  SECCIÓN 3: FUNCIONES DE LÓGICA
    // ════════════════════════════════════════════
    static void Sumar()
    {
        Console.Write("\nIngresa el primer número: ");
        double a = double.Parse(Console.ReadLine());

        Console.Write("Ingresa el segundo número: ");
        double b = double.Parse(Console.ReadLine());

        Console.WriteLine($"\n✅ Resultado: {a} + {b} = {a + b}");
    }

    static void Restar()
    {
        Console.Write("\nIngresa el primer número: ");
        double a = double.Parse(Console.ReadLine());

        Console.Write("Ingresa el segundo número: ");
        double b = double.Parse(Console.ReadLine());

        Console.WriteLine($"\n✅ Resultado: {a} - {b} = {a - b}");
    }

    static void Multiplicar()
    {
        Console.Write("\nIngresa el primer número: ");
        double a = double.Parse(Console.ReadLine());

        Console.Write("Ingresa el segundo número: ");
        double b = double.Parse(Console.ReadLine());

        Console.WriteLine($"\n✅ Resultado: {a} × {b} = {a * b}");
    }

    // ════════════════════════════════════════════
    //  SECCIÓN 4: FUNCIONES DE UTILIDAD
    // ════════════════════════════════════════════
    static void Pausa()
    {
        Console.Write("\nPresiona ENTER para continuar...");
        Console.ReadLine();
    }

    static void MensajeSalida()
    {
        Console.WriteLine("\n👋 ¡Hasta pronto!");
    }

    static void MensajeError()
    {
        Console.WriteLine("\n⚠️  Opción no válida. Intenta de nuevo.");
    }
}

Bloque 2 — Menús con Funciones Avanzadas

Retornos, parámetros y submenús
⏱️ 60 min 📚 3 conceptos

↩️ Concepto 2.1 — Funciones que retornan valores

Hasta ahora, nuestras funciones hacen cosas pero no devuelven resultados. Las funciones con return son más poderosas porque pueden usarse dentro de cálculos o para tomar decisiones.

Void vs Return 
// Función que NO retorna (void) — solo hace algo
static void Saludar()
{
    Console.WriteLine("Hola");
}

// Función que SÍ retorna — produce un resultado
static double Calcular(double a, double b)
{
    return a + b;   // ← devuelve el valor
}

📥 Concepto 2.2 — Funciones con parámetros

Los parámetros son los datos que le enviamos a una función para que trabaje con ellos.

Parámetros y argumentos 
//          parámetros (declaración)
//          ↓       ↓
static double Sumar(double a, double b)
{
    return a + b;    // ← retorno
}

// Forma de llamarla:
double resultado = Sumar(5.0, 3.0);
//                       ↑     ↑
//               argumentos (valores reales)

📝 Ejemplo 3 — Sistema de notas con funciones que retornan

SistemaNotas.cs 
using System;

class SistemaNotas
{
    static void Main()
    {
        string opcion;

        do
        {
            MostrarMenu();
            opcion = Console.ReadLine();

            switch (opcion)
            {
                case "1": FlujoCalcularPromedio(); break;
                case "2": FlujoClasificarNota();   break;
                case "3": FlujoEstaAprobado();     break;
                case "0": Console.WriteLine("\n👋 ¡Adiós!"); break;
                default:  Console.WriteLine("\n⚠️  Opción inválida."); break;
            }

            if (opcion != "0") Pausa();

        } while (opcion != "0");
    }

    // ─── MENÚ ─────────────────────────────────────────
    static void MostrarMenu()
    {
        Console.Clear();
        Console.WriteLine("┌────────────────────────────┐");
        Console.WriteLine("│     SISTEMA DE NOTAS       │");
        Console.WriteLine("├────────────────────────────┤");
        Console.WriteLine("│  1. Calcular promedio      │");
        Console.WriteLine("│  2. Clasificar una nota    │");
        Console.WriteLine("│  3. ¿Está aprobado?        │");
        Console.WriteLine("│  0. Salir                  │");
        Console.WriteLine("└────────────────────────────┘");
        Console.Write("  Opción: ");
    }

    // ─── FLUJOS (conectan menú con funciones lógicas) ─
    static void FlujoCalcularPromedio()
    {
        Console.Write("\n¿Cuántas notas vas a ingresar? ");
        int cantidad = int.Parse(Console.ReadLine());
        double[] notas = new double[cantidad];

        for (int i = 0; i < cantidad; i++)
        {
            Console.Write($"  Nota {i + 1}: ");
            notas[i] = double.Parse(Console.ReadLine());
        }

        double promedio = CalcularPromedio(notas);
        Console.WriteLine($"\n📊 El promedio es: {promedio:F2}");
    }

    static void FlujoClasificarNota()
    {
        Console.Write("\nIngresa una nota (0-100): ");
        double nota = double.Parse(Console.ReadLine());
        string clasificacion = ClasificarNota(nota);
        Console.WriteLine($"\n🏅 Clasificación: {clasificacion}");
    }

    static void FlujoEstaAprobado()
    {
        Console.Write("\nIngresa la nota final: ");
        double nota = double.Parse(Console.ReadLine());
        bool aprobado = EstaAprobado(nota);
        string mensaje = aprobado ? "✅ APROBADO" : "❌ REPROBADO";
        Console.WriteLine($"\nEstado: {mensaje}");
    }

    // ─── FUNCIONES LÓGICAS (retornan valores) ────────
    static double CalcularPromedio(double[] notas)
    {
        double suma = 0;
        foreach (double nota in notas) suma += nota;
        return suma / notas.Length;
    }

    static string ClasificarNota(double nota)
    {
        if      (nota >= 90) return "⭐ Excelente";
        else if (nota >= 75) return "👍 Bueno";
        else if (nota >= 60) return "😐 Regular";
        else                 return "😟 Insuficiente";
    }

    static bool EstaAprobado(double nota)
    {
        return nota >= 60;
    }

    static void Pausa()
    {
        Console.Write("\nPresiona ENTER para continuar...");
        Console.ReadLine();
    }
}
💡
Observa los tres tipos de retorno
  • CalcularPromedio recibe un arreglo y devuelve un double.
  • ClasificarNota recibe un double y devuelve un string.
  • EstaAprobado recibe un double y devuelve un bool.
  • Los flujos son funciones intermedias que leen datos y luego llaman a las funciones lógicas.

using System;

class MenuConSubmenu
{
    static void Main()
    {
        string opcion;

        do
        {
            Console.Clear();
            Console.WriteLine("═══════════════════════════════");
            Console.WriteLine("       MENÚ PRINCIPAL          ");
            Console.WriteLine("═══════════════════════════════");
            Console.WriteLine("  1. Módulo de Matemáticas");
            Console.WriteLine("  2. Módulo de Texto");

    // ─── SUBMENÚ 1: Matemáticas ───────────────────
    static void MenuMatematicas()
    {
        string opcion;
        do
        {
            Console.Clear();
            Console.WriteLine("  ┌─────────────────────────┐");
            Console.WriteLine("  │   MÓDULO MATEMÁTICAS    │");
            Console.WriteLine("  ├─────────────────────────┤");
            Console.WriteLine("  │  1. Números pares/impares│");
            Console.WriteLine("  │  2. Tabla de multiplicar │");
            Console.WriteLine("  │  3. Factorial            │");
            Console.WriteLine("  │  0. ← Volver             │");
            Console.WriteLine("  └─────────────────────────┘");
            Console.Write("  Opción: ");
            opcion = Console.ReadLine();

            switch (opcion)
            {
                case "1": MostrarParesImpares(); break;
                case "2": MostrarTabla();        break;
                case "3": MostrarFactorial();    break;
                case "0": break;
                default:  Console.WriteLine("⚠️ Inválido."); break;
            }
            if (opcion != "0") Pausa();
        } while (opcion != "0");
    }

    // ─── SUBMENÚ 2: Texto ─────────────────────────
    static void MenuTexto()
    {
        string opcion;
        do
        {
            Console.Clear();
            Console.WriteLine("  ┌─────────────────────────┐");
            Console.WriteLine("  │      MÓDULO TEXTO       │");
            Console.WriteLine("  ├─────────────────────────┤");
            Console.WriteLine("  │  1. Mayúsculas/Minúsculas│");
            Console.WriteLine("  │  2. Contar caracteres   │");
            Console.WriteLine("  │  3. Invertir texto      │");
            Console.WriteLine("  │  0. ← Volver            │");
            Console.WriteLine("  └─────────────────────────┘");
            Console.Write("  Opción: ");
            opcion = Console.ReadLine();

            switch (opcion)
            {
                case "1": CambiarMayusculas();  break;
                case "2": ContarCaracteres();   break;
                case "3": InvertirTexto();      break;
                case "0": break;
                default:  Console.WriteLine("⚠️ Inválido."); break;
            }
            if (opcion != "0") Pausa();
        } while (opcion != "0");
    }

    // ─── FUNCIONES DE MATEMÁTICAS ─────────────────
    static void MostrarParesImpares()
    {
        Console.Write("\nHasta qué número? ");
        int n = int.Parse(Console.ReadLine());

        Console.WriteLine("\nPares:");
        for (int i = 2; i <= n; i += 2) Console.Write(i + " ");

        Console.WriteLine("\nImpares:");
        for (int i = 1; i <= n; i += 2) Console.Write(i + " ");
        Console.WriteLine();
    }

    static void MostrarTabla()
    {
        Console.Write("\n¿Tabla del número? ");
        int n = int.Parse(Console.ReadLine());
        Console.WriteLine($"\n── Tabla del {n} ──");
        for (int i = 1; i <= 10; i++)
            Console.WriteLine($"  {n} × {i,2} = {n * i,3}");
    }

    static void MostrarFactorial()
    {
        Console.Write("\n¿Factorial de qué número? ");
        int n = int.Parse(Console.ReadLine());
        long resultado = Factorial(n);
        Console.WriteLine($"\n  {n}! = {resultado}");
    }

    static long Factorial(int n)
    {
        if (n <= 1) return 1;
        return n * Factorial(n - 1);   // función recursiva
    }

    // ─── FUNCIONES DE TEXTO ───────────────────────
    static void CambiarMayusculas()
    {
        Console.Write("\nEscribe un texto: ");
        string texto = Console.ReadLine();
        Console.WriteLine($"\nMayúsculas: {texto.ToUpper()}");
        Console.WriteLine($"Minúsculas: {texto.ToLower()}");
    }

    static void ContarCaracteres()
    {
        Console.Write("\nEscribe un texto: ");
        string texto = Console.ReadLine();
        Console.WriteLine($"\nTotal de caracteres (con espacios): {texto.Length}");
        Console.WriteLine($"Total sin espacios: {texto.Replace(" ", "").Length}");
    }

    static void InvertirTexto()
    {
        Console.Write("\nEscribe un texto: ");
        string texto = Console.ReadLine();
        char[] caracteres = texto.ToCharArray();
        Array.Reverse(caracteres);
        string invertido = new string(caracteres);
        Console.WriteLine($"\nTexto invertido: {invertido}");
    }

    static void Pausa()
    {
        Console.Write("\nPresiona ENTER para continuar...");
        Console.ReadLine();
    }
}

Bloque 3 — Ciclos, Condiciones y Menús

Programas robustos que no se rompen
⏱️ 50 min 📚 2 conceptos

🛡️ Concepto 3.1 — Validación de entradas con ciclos

Un programa robusto no debe caerse si el usuario escribe algo inesperado. Usa ciclos while para obligar al usuario a ingresar datos válidos.

❌ Sin validación
Console.Write("Número: ");
int n = int.Parse(
  Console.ReadLine());
// 💥 Se rompe con "abc"
✅ Con validación
int n;
while(!int.TryParse(
  Console.ReadLine(),
  out n)) {
  // Sigue pidiendo
}

🎯 Concepto 3.2 — Validación con rango

Validaciones reutilizables 
// Lee un entero cualquiera, con reintentos
static int LeerEntero(string mensaje)
{
    int numero;
    bool valido;

    do
    {
        Console.Write(mensaje);
        valido = int.TryParse(Console.ReadLine(), out numero);
        if (!valido)
            Console.WriteLine("⚠️  Por favor, ingresa solo números enteros.");
    } while (!valido);

    return numero;
}

// Lee un entero dentro de un rango [minimo, maximo]
static int LeerEnteroEnRango(string mensaje, int minimo, int maximo)
{
    int numero;
    bool valido;

    do
    {
        Console.Write(mensaje);
        valido = int.TryParse(Console.ReadLine(), out numero);

        if (!valido)
            Console.WriteLine("⚠️  Debe ser un número.");
        else if (numero < minimo || numero > maximo)
        {
            Console.WriteLine($"⚠️  Debe estar entre {minimo} y {maximo}.");
            valido = false;
        }
    } while (!valido);

    return numero;
}

📝 Ejemplo 5 — Programa completo con validaciones

ProgramaConValidaciones.cs 
using System;

class ProgramaConValidaciones
{
    static void Main()
    {
        int opcion;
        do
        {
            MostrarMenu();
            opcion = LeerEnteroEnRango("  Opción: ", 0, 3);

            switch (opcion)
            {
                case 1: JuegoAdivinanza();       break;
                case 2: ContadorPersonalizado(); break;
                case 3: AnalisisNumeros();       break;
                case 0: Console.WriteLine("\n👋 ¡Hasta luego!"); break;
            }
            if (opcion != 0) Pausa();
        } while (opcion != 0);
    }

    static void MostrarMenu()
    {
        Console.Clear();
        Console.WriteLine("╔══════════════════════════════╗");
        Console.WriteLine("║    JUEGOS Y HERRAMIENTAS     ║");
        Console.WriteLine("╠══════════════════════════════╣");
        Console.WriteLine("║  1. Juego: Adivina el número ║");
        Console.WriteLine("║  2. Contador personalizado   ║");
        Console.WriteLine("║  3. Análisis de números      ║");
        Console.WriteLine("║  0. Salir                    ║");
        Console.WriteLine("╚══════════════════════════════╝");
    }

    static void JuegoAdivinanza()
    {
        Random rnd = new Random();
        int secreto = rnd.Next(1, 101);
        int intentos = 0, maximo = 7, intento;

        Console.WriteLine($"\n🎮 Adivina el número (1-100). Tienes {maximo} intentos.");

        do
        {
            intento = LeerEnteroEnRango($"\n  Intento {intentos + 1}/{maximo}: ", 1, 100);
            intentos++;

            if      (intento < secreto) Console.WriteLine("  📈 Más alto");
            else if (intento > secreto) Console.WriteLine("  📉 Más bajo");
            else                        Console.WriteLine($"\n  🎉 ¡Correcto en {intentos} intento(s)!");
        } while (intento != secreto && intentos < maximo);

        if (intento != secreto)
            Console.WriteLine($"\n  😢 Se acabaron los intentos. Era el {secreto}.");
    }

    static void ContadorPersonalizado()
    {
        Console.WriteLine("\n📟 CONTADOR PERSONALIZADO");
        int inicio = LeerEntero("\n  Número de inicio: ");
        int fin    = LeerEntero("  Número de fin:     ");
        int paso   = LeerEntero("  Paso (incremento): ");

        if (paso == 0)
        {
            Console.WriteLine("⚠️  El paso no puede ser cero.");
            return;
        }

        Console.WriteLine("\n  Contando:");
        Console.Write("   ");
        if (paso > 0) {
            for (int i = inicio; i <= fin; i += paso) Console.Write(i + " ");
        } else {
            for (int i = inicio; i >= fin; i += paso) Console.Write(i + " ");
        }
        Console.WriteLine();
    }

    static void AnalisisNumeros()
    {
        int cantidad = LeerEnteroEnRango("\n¿Cuántos números analizarás? (1-20): ", 1, 20);
        int suma = 0, mayor = int.MinValue, menor = int.MaxValue, pares = 0, impares = 0;

        for (int i = 1; i <= cantidad; i++)
        {
            int num = LeerEntero($"  Número {i}: ");
            suma += num;
            if (num > mayor) mayor = num;
            if (num < menor) menor = num;
            if (num % 2 == 0) pares++; else impares++;
        }

        Console.WriteLine("\n  ── RESULTADOS ──────────────");
        Console.WriteLine($"  Suma:     {suma}");
        Console.WriteLine($"  Promedio: {(double)suma / cantidad:F2}");
        Console.WriteLine($"  Mayor:    {mayor}");
        Console.WriteLine($"  Menor:    {menor}");
        Console.WriteLine($"  Pares:    {pares}");
        Console.WriteLine($"  Impares:  {impares}");
    }

    static int LeerEntero(string mensaje)
    {
        int numero;
        while (true)
        {
            Console.Write(mensaje);
            if (int.TryParse(Console.ReadLine(), out numero))
                return numero;
            Console.WriteLine("  ⚠️  Solo números enteros.");
        }
    }

    static int LeerEnteroEnRango(string mensaje, int minimo, int maximo)
    {
        int numero;
        while (true)
        {
            Console.Write(mensaje);
            if (int.TryParse(Console.ReadLine(), out numero) &&
                numero >= minimo && numero <= maximo)
                return numero;
            Console.WriteLine($"  ⚠️  Ingresa un número entre {minimo} y {maximo}.");
        }
    }

    static void Pausa()
    {
        Console.Write("\nPresiona ENTER para continuar...");
        Console.ReadLine();
    }
}

Bloque 4 — Proyecto Integrador

Sistema de Registro de Estudiantes
⏱️ 40 min 🎯 Proyecto completo

Este proyecto integra todo lo aprendido: menú principal, submenús, funciones, validaciones, ciclos y condiciones.

SistemaEstudiantes.cs 
using System;

class SistemaEstudiantes
{
    // ════════════════════════════════════════════
    //  DATOS GLOBALES (arreglos paralelos)
    // ════════════════════════════════════════════
    const int MAX = 20;
    static string[] nombres   = new string[MAX];
    static double[] promedios = new double[MAX];
    static int totalEstudiantes = 0;

    static void Main()
    {
        int opcion;
        do
        {
            MostrarMenuPrincipal();
            opcion = LeerEnteroEnRango("  Opción: ", 0, 4);

            switch (opcion)
            {
                case 1: AgregarEstudiante();   break;
                case 2: MostrarEstudiantes();  break;
                case 3: MenuBusqueda();        break;
                case 4: MostrarEstadisticas(); break;
                case 0: Console.WriteLine("\n👋 Sistema cerrado."); break;
            }
            if (opcion != 0) Pausa();
        } while (opcion != 0);
    }

    static void MostrarMenuPrincipal()
    {
        Console.Clear();
        Console.WriteLine("╔═══════════════════════════════════╗");
        Console.WriteLine("║    REGISTRO DE ESTUDIANTES v1.0   ║");
        Console.WriteLine($"║         Registros: {totalEstudiantes,3}/{MAX}            ║");
        Console.WriteLine("╠═══════════════════════════════════╣");
        Console.WriteLine("║  1. Agregar estudiante            ║");
        Console.WriteLine("║  2. Ver todos los estudiantes     ║");
        Console.WriteLine("║  3. Buscar estudiante             ║");
        Console.WriteLine("║  4. Ver estadísticas              ║");
        Console.WriteLine("║  0. Salir                         ║");
        Console.WriteLine("╚═══════════════════════════════════╝");
    }

    static void MenuBusqueda()
    {
        if (totalEstudiantes == 0)
        {
            Console.WriteLine("\n⚠️  No hay estudiantes registrados.");
            return;
        }

        string opcion;
        do
        {
            Console.Clear();
            Console.WriteLine("  ┌────────────────────────┐");
            Console.WriteLine("  │      BÚSQUEDA          │");
            Console.WriteLine("  ├────────────────────────┤");
            Console.WriteLine("  │  1. Por nombre         │");
            Console.WriteLine("  │  2. Mejor promedio     │");
            Console.WriteLine("  │  3. Por estado         │");
            Console.WriteLine("  │  0. Volver             │");
            Console.WriteLine("  └────────────────────────┘");
            Console.Write("  Opción: ");
            opcion = Console.ReadLine();

            switch (opcion)
            {
                case "1": BuscarPorNombre();     break;
                case "2": MostrarMejorPromedio(); break;
                case "3": BuscarPorEstado();     break;
                case "0": break;
                default: Console.WriteLine("⚠️ Inválido."); break;
            }
            if (opcion != "0") Pausa();
        } while (opcion != "0");
    }

    static void AgregarEstudiante()
    {
        if (totalEstudiantes >= MAX)
        {
            Console.WriteLine("\n⚠️  El sistema está lleno.");
            return;
        }

        Console.WriteLine("\n── Agregar Estudiante ──────────────");
        Console.Write("Nombre: ");
        string nombre = Console.ReadLine().Trim();

        if (string.IsNullOrEmpty(nombre))
        {
            Console.WriteLine("⚠️  El nombre no puede estar vacío.");
            return;
        }

        double promedio = LeerDecimalEnRango("Promedio (0.0 - 100.0): ", 0.0, 100.0);
        nombres[totalEstudiantes]   = nombre;
        promedios[totalEstudiantes] = promedio;
        totalEstudiantes++;
        Console.WriteLine($"\n✅ '{nombre}' registrado con promedio {promedio:F1}");
    }

    static void MostrarEstudiantes()
    {
        Console.WriteLine("\n── Lista de Estudiantes ────────────");
        if (totalEstudiantes == 0) { Console.WriteLine("  (Sin registros)"); return; }

        Console.WriteLine($"  {"#",-4} {"Nombre",-20} {"Promedio",8}  Estado");
        Console.WriteLine("   " + new string('─', 45));
        for (int i = 0; i < totalEstudiantes; i++)
        {
            string estado = ObtenerEstado(promedios[i]);
            Console.WriteLine($"  {i + 1,-4} {nombres[i],-20} {promedios[i],8:F1}  {estado}");
        }
    }

    static void BuscarPorNombre()
    {
        Console.Write("\nNombre a buscar: ");
        string busqueda = Console.ReadLine().ToLower();
        bool encontrado = false;

        for (int i = 0; i < totalEstudiantes; i++)
        {
            if (nombres[i].ToLower().Contains(busqueda))
            {
                Console.WriteLine($"\n  → {nombres[i]}: {promedios[i]:F1} ({ObtenerEstado(promedios[i])})");
                encontrado = true;
            }
        }
        if (!encontrado) Console.WriteLine("\n  No se encontraron coincidencias.");
    }

    static void MostrarMejorPromedio()
    {
        int indiceMejor = 0;
        for (int i = 1; i < totalEstudiantes; i++)
            if (promedios[i] > promedios[indiceMejor]) indiceMejor = i;
        Console.WriteLine($"\n🏆 Mejor promedio: {nombres[indiceMejor]} con {promedios[indiceMejor]:F1}");
    }

    static void BuscarPorEstado()
    {
        Console.WriteLine("\n  Estados: 1=Excelente  2=Bueno  3=Regular  4=Insuficiente");
        Console.Write("  ¿Qué estado buscar? ");
        string estadoBuscar = ObtenerEstadoPorOpcion(Console.ReadLine());
        bool encontrado = false;
        Console.WriteLine($"\n  Estudiantes en estado '{estadoBuscar}':");

        for (int i = 0; i < totalEstudiantes; i++)
        {
            if (ObtenerEstado(promedios[i]) == estadoBuscar)
            {
                Console.WriteLine($"  → {nombres[i]}: {promedios[i]:F1}");
                encontrado = true;
            }
        }
        if (!encontrado) Console.WriteLine("  (Ninguno)");
    }

    static void MostrarEstadisticas()
    {
        if (totalEstudiantes == 0) { Console.WriteLine("\n⚠️  No hay datos."); return; }

        double suma = 0, mayor = promedios[0], menor = promedios[0];
        int aprobados = 0;

        for (int i = 0; i < totalEstudiantes; i++)
        {
            suma += promedios[i];
            if (promedios[i] > mayor) mayor = promedios[i];
            if (promedios[i] < menor) menor = promedios[i];
            if (promedios[i] >= 60) aprobados++;
        }

        double promGeneral = suma / totalEstudiantes;
        Console.WriteLine("\n── Estadísticas del Grupo ──────────");
        Console.WriteLine($"  Total estudiantes:  {totalEstudiantes}");
        Console.WriteLine($"  Promedio general:   {promGeneral:F2}");
        Console.WriteLine($"  Mayor promedio:     {mayor:F1}");
        Console.WriteLine($"  Menor promedio:     {menor:F1}");
        Console.WriteLine($"  Aprobados:          {aprobados} ({(double)aprobados / totalEstudiantes * 100:F0}%)");
        Console.WriteLine($"  Reprobados:         {totalEstudiantes - aprobados}");
    }

    // ════════════════════════════════════════════
    //  FUNCIONES DE APOYO
    // ════════════════════════════════════════════
    static string ObtenerEstado(double promedio)
    {
        if      (promedio >= 90) return "Excelente";
        else if (promedio >= 75) return "Bueno";
        else if (promedio >= 60) return "Regular";
        else                     return "Insuficiente";
    }

    static string ObtenerEstadoPorOpcion(string op)
    {
        switch (op)
        {
            case "1": return "Excelente";
            case "2": return "Bueno";
            case "3": return "Regular";
            case "4": return "Insuficiente";
            default:  return "Desconocido";
        }
    }

    static int LeerEnteroEnRango(string msg, int min, int max)
    {
        int n;
        while (true)
        {
            Console.Write(msg);
            if (int.TryParse(Console.ReadLine(), out n) && n >= min && n <= max)
                return n;
            Console.WriteLine($"  ⚠️  Ingresa un número entre {min} y {max}.");
        }
    }

    static double LeerDecimalEnRango(string msg, double min, double max)
    {
        double n;
        while (true)
        {
            Console.Write(msg);
            if (double.TryParse(Console.ReadLine(), out n) && n >= min && n <= max)
                return n;
            Console.WriteLine($"  ⚠️  Ingresa un número entre {min} y {max}.");
        }
    }

    static void Pausa()
    {
        Console.Write("\nPresiona ENTER para continuar...");
        Console.ReadLine();
    }
}
